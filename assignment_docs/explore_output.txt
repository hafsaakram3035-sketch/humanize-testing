=== intcomma ===
intcomma(0,)                                            -> '0'
intcomma(1,)                                            -> '1'
intcomma(999,)                                          -> '999'
intcomma(1000,)                                         -> '1,000'
intcomma(1001,)                                         -> '1,001'
intcomma(12345,)                                        -> '12,345'
intcomma(-999,)                                         -> '-999'
intcomma(-1000,)                                        -> '-1,000'
intcomma(1000.5,)                                       -> '1,000.5'
intcomma('1000',)                                       -> '1,000'
intcomma('abc',)                                        -> 'abc'
intcomma('',)                                           -> ''
intcomma(None,)                                         -> None
intcomma(1234.5678, 2)                                  -> '1,234.57'

=== naturalsize ===
naturalsize(0,)                                         -> '0 Bytes'
naturalsize(1,)                                         -> '1 Byte'
naturalsize(2,)                                         -> '2 Bytes'
naturalsize(999,)                                       -> '999 Bytes'
naturalsize(1000,)                                      -> '1.0 kB'
naturalsize(1001,)                                      -> '1.0 kB'
naturalsize(1500,)                                      -> '1.5 kB'
naturalsize(1000000,)                                   -> '1.0 MB'
naturalsize(1000000000,)                                -> '1.0 GB'
naturalsize(-1000,)                                     -> '-1.0 kB'
naturalsize('1000',)                                    -> '1.0 kB'
naturalsize('abc',)                                     -> ERROR ValueError: could not convert string to float: 'abc'
naturalsize(None,)                                      -> ERROR TypeError: float() argument must be a string or a real number, not 'NoneType'
naturalsize(1023,){'binary': True}                      -> '1023 Bytes'
naturalsize(1024,){'binary': True}                      -> '1.0 KiB'
naturalsize(1000,){'gnu': True}                         -> '1000B'
naturalsize(1024,){'gnu': True}                         -> '1.0K'
naturalsize(1500,){'format': '%.2f'}                    -> '1.50 kB'
